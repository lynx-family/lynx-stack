## 类 \{#classes\}

### `ReactRefreshRspackPlugin` \{#reactrefreshrspackplugin\}

ReactRefreshRspackPlugin 用于在 ReactLynx 中启用 fast refresh。

```js
// rspack.config.js
import { ReactRefreshRspackPlugin } from '@lynx-js/react-refresh-webpack-plugin'
export default {
  module: {
    rules: [
      {
        test: /\.[jt]sx?$/,
        exclude: [
          /node_modules/,
          /@lynx-js/,
          /react-refresh/,
          /compiler-nodiff-runtime3/,
          ReactRefreshRspackPlugin.loader,
        ],
        use: [ReactRefreshRspackPlugin.loader],
      },
    ],
  },
  plugins: [new ReactRefreshRspackPlugin()],
}
```

#### `ReactRefreshRspackPlugin.loader` \{#reactrefreshrspackplugin-loader\}

- **类型：** `string`

React Refresh 的 loader。

注意需要忽略 React 的运行时，参见下方示例：

```js
// rspack.config.js
import { ReactRefreshRspackPlugin } from '@lynx-js/react-refresh-webpack-plugin'
import { ReactWebpackPlugin } from '@lynx-js/react-webpack-plugin'

export default {
  module: {
    rules: [
      {
        issueLayer: ReactWebpackPlugin.LAYERS.BACKGROUND,
        test: /\.[jt]sx?$/,
        exclude: [
          /node_modules/,
          /@lynx-js/react/,
          ReactRefreshRspackPlugin.loader,
        ],
        use: [ReactRefreshRspackPlugin.loader],
      },
    ],
  },
  plugins: [new ReactRefreshRspackPlugin()],
}
```
